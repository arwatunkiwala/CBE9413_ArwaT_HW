using JuMP
using Gurobi
using CSV
using DataFrames

# Read time series data
data = CSV.read("demand_vre_cf.csv", DataFrame)

# Number of time periods
T = nrow(data)  # 8760

# Extract time series
load = data.load
wind_cf = data.onwind
solar_cf = data.solar

# Parameters - Economic
discount_rate = 0.07
lifetime = 30
CRF = (discount_rate * (1 + discount_rate)^lifetime) / ((1 + discount_rate)^lifetime - 1)

# Generator parameters
wind_capex = 1335; wind_fom = 28; wind_vom = 0
wind_ann_cost = (wind_capex * CRF + wind_fom) * 1000

solar_capex = 895; solar_fom = 15; solar_vom = 0
solar_ann_cost = (solar_capex * CRF + solar_fom) * 1000

gas_capex = 1343; gas_fom = 30; gas_vom = 2
heat_rate = 6.16; gas_price = 4
gas_fuel_cost = heat_rate * gas_price
gas_total_vom = gas_vom + gas_fuel_cost
gas_ann_cost = (gas_capex * CRF + gas_fom) * 1000
gas_co2_intensity = heat_rate * 0.053

# Storage parameters
liion_capex_power = 257; liion_fom_power = 6.4
liion_capex_energy = 277; liion_fom_energy = 6.9; liion_vom = 1
liion_ann_cost_power = (liion_capex_power * CRF + liion_fom_power) * 1000
liion_ann_cost_energy = (liion_capex_energy * CRF + liion_fom_energy) * 1000
liion_charge_eff = 0.92; liion_discharge_eff = 0.92
liion_self_discharge = 0.001
liion_min_soc = 0.05; liion_max_soc = 0.95
liion_min_duration = 1; liion_max_duration = 10

metalair_capex_power = 1100; metalair_fom_power = 27.5
metalair_capex_energy = 10; metalair_fom_energy = 0.25; metalair_vom = 1
metalair_ann_cost_power = (metalair_capex_power * CRF + metalair_fom_power) * 1000
metalair_ann_cost_energy = (metalair_capex_energy * CRF + metalair_fom_energy) * 1000
metalair_charge_eff = 0.72; metalair_discharge_eff = 0.59
metalair_self_discharge = 0.01
metalair_min_soc = 0.05; metalair_max_soc = 0.95
metalair_min_duration = 10; metalair_max_duration = 150

load_shed_cost = 5000

# =============================================================================
# PART F: FIX CAPACITIES FROM PART C AND PART D, TEST IN FULL MODEL
# =============================================================================

# Capacities from Part C (3-hour aggregation)
C_wind_partC = 3114.23
C_solar_partC = 3028.36
C_gas_partC = 1948.77
C_liion_P_partC = 0.0
C_liion_E_partC = 0.0
C_metalair_P_partC = 697.72
C_metalair_E_partC = 18322.68

# Capacities from Part D K=10
C_wind_partD10 = 2818.85
C_solar_partD10 = 2910.44
C_gas_partD10 = 1772.30
C_liion_P_partD10 = 452.94
C_liion_E_partD10 = 1304.41
C_metalair_P_partD10 = 84.47
C_metalair_E_partD10 = 1052.15

# Capacities from Part D K=20
C_wind_partD20 = 2935.95
C_solar_partD20 = 2933.18
C_gas_partD20 = 1803.29
C_liion_P_partD20 = 412.51
C_liion_E_partD20 = 1164.78
C_metalair_P_partD20 = 123.37
C_metalair_E_partD20 = 1449.18

# Function to run validation
function validate_capacities(name, C_wind_fix, C_solar_fix, C_gas_fix, 
                             C_liion_P_fix, C_liion_E_fix, 
                             C_metalair_P_fix, C_metalair_E_fix)
    
    println("\n" * "="^70)
    println("VALIDATING: $name")
    println("="^70)
    
    # Create model
    model = Model(Gurobi.Optimizer)
    set_optimizer_attribute(model, "OutputFlag", 0)  # Suppress Gurobi output
    
    # Operational variables only (capacities are fixed)
    @variable(model, y_wind[1:T] >= 0)
    @variable(model, y_solar[1:T] >= 0)
    @variable(model, y_gas[1:T] >= 0)
    @variable(model, PC_liion[1:T] >= 0)
    @variable(model, PD_liion[1:T] >= 0)
    @variable(model, S_liion[1:T] >= 0)
    @variable(model, PC_metalair[1:T] >= 0)
    @variable(model, PD_metalair[1:T] >= 0)
    @variable(model, S_metalair[1:T] >= 0)
    @variable(model, LS[1:T] >= 0)
    
    # Objective: Minimize operational costs only (no capital costs)
    @objective(model, Min,
        sum(wind_vom * y_wind[t] + 
            solar_vom * y_solar[t] + 
            gas_total_vom * y_gas[t] +
            liion_vom * (PC_liion[t] + PD_liion[t]) +
            metalair_vom * (PC_metalair[t] + PD_metalair[t]) +
            load_shed_cost * LS[t]
            for t in 1:T)
    )
    
    # Constraints
    
    # 1. Supply-demand balance
    @constraint(model, balance[t in 1:T],
        y_wind[t] + y_solar[t] + y_gas[t] + 
        PD_liion[t] + PD_metalair[t] - 
        PC_liion[t] - PC_metalair[t] + 
        LS[t] == load[t]
    )
    
    # 2. Generation capacity limits (FIXED CAPACITIES)
    @constraint(model, wind_cap[t in 1:T],
        y_wind[t] <= wind_cf[t] * C_wind_fix
    )
    
    @constraint(model, solar_cap[t in 1:T],
        y_solar[t] <= solar_cf[t] * C_solar_fix
    )
    
    @constraint(model, gas_cap[t in 1:T],
        y_gas[t] <= C_gas_fix
    )
    
    # 3. Storage power limits (FIXED CAPACITIES)
    @constraint(model, liion_power[t in 1:T],
        PC_liion[t] + PD_liion[t] <= C_liion_P_fix
    )
    
    @constraint(model, metalair_power[t in 1:T],
        PC_metalair[t] + PD_metalair[t] <= C_metalair_P_fix
    )
    
    # 4. Storage SOC limits (FIXED CAPACITIES)
    @constraint(model, liion_soc_min[t in 1:T],
        S_liion[t] >= liion_min_soc * C_liion_E_fix
    )
    
    @constraint(model, liion_soc_max[t in 1:T],
        S_liion[t] <= liion_max_soc * C_liion_E_fix
    )
    
    @constraint(model, metalair_soc_min[t in 1:T],
        S_metalair[t] >= metalair_min_soc * C_metalair_E_fix
    )
    
    @constraint(model, metalair_soc_max[t in 1:T],
        S_metalair[t] <= metalair_max_soc * C_metalair_E_fix
    )
    
    # 5. Storage inventory balance
    @constraint(model, liion_balance_first,
        S_liion[1] == (1 - liion_self_discharge) * S_liion[T] + 
                       liion_charge_eff * PC_liion[1] - 
                       PD_liion[1] / liion_discharge_eff
    )
    
    @constraint(model, liion_balance[t in 2:T],
        S_liion[t] == (1 - liion_self_discharge) * S_liion[t-1] + 
                       liion_charge_eff * PC_liion[t] - 
                       PD_liion[t] / liion_discharge_eff
    )
    
    @constraint(model, metalair_balance_first,
        S_metalair[1] == (1 - metalair_self_discharge) * S_metalair[T] + 
                          metalair_charge_eff * PC_metalair[1] - 
                          PD_metalair[1] / metalair_discharge_eff
    )
    
    @constraint(model, metalair_balance[t in 2:T],
        S_metalair[t] == (1 - metalair_self_discharge) * S_metalair[t-1] + 
                          metalair_charge_eff * PC_metalair[t] - 
                          PD_metalair[t] / metalair_discharge_eff
    )
    
    # NOTE: NO EMISSIONS CONSTRAINT (as per bonus instructions)
    
    # Solve
    optimize!(model)
    
    # Extract results
    status = termination_status(model)
    
    if status == MOI.OPTIMAL
        total_load_shed = sum(value.(LS))
        total_demand = sum(load)
        load_shed_pct = 100 * total_load_shed / total_demand
        
        gas_gen = sum(value.(y_gas))
        total_emissions = gas_co2_intensity * gas_gen
        
        println("\nRESULTS:")
        println("Status: OPTIMAL")
        println("\nFixed Capacities:")
        println("  Wind: $(round(C_wind_fix, digits=2)) MW")
        println("  Solar: $(round(C_solar_fix, digits=2)) MW")
        println("  Gas: $(round(C_gas_fix, digits=2)) MW")
        println("  Li-ion: $(round(C_liion_P_fix, digits=2)) MW / $(round(C_liion_E_fix, digits=2)) MWh")
        println("  Metal-air: $(round(C_metalair_P_fix, digits=2)) MW / $(round(C_metalair_E_fix, digits=2)) MWh")
        
        println("\nOperational Results:")
        println("  Annual Load Shedding: $(round(total_load_shed, digits=2)) MWh")
        println("  Load Shedding as % of Demand: $(round(load_shed_pct, digits=4))%")
        println("  Annual CO2 Emissions: $(round(total_emissions, digits=2)) tCO2")
        println("  Emissions as % of 1M budget: $(round(100*total_emissions/1e6, digits=2))%")
        
        return Dict(
            "name" => name,
            "status" => "OPTIMAL",
            "load_shed_MWh" => total_load_shed,
            "load_shed_pct" => load_shed_pct,
            "emissions_tCO2" => total_emissions,
            "emissions_pct" => 100*total_emissions/1e6
        )
    else
        println("\nRESULTS:")
        println("Status: $status (INFEASIBLE OR ERROR)")
        return Dict(
            "name" => name,
            "status" => string(status),
            "load_shed_MWh" => NaN,
            "load_shed_pct" => NaN,
            "emissions_tCO2" => NaN,
            "emissions_pct" => NaN
        )
    end
end

# Run validations
